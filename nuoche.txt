require("dotenv").config();


const express=require("express");

const axios=require("axios");

const nodemailer=require("nodemailer");

const fs=require("fs");

const path=require("path");

const bodyParser=require("body-parser");

const {v4:uuidv4}=require("uuid");



const app=express();



const PORT=process.env.PORT || 5000;



const DB_FILE=path.join(
__dirname,
"data/orders.json"
);





app.use(
express.static("public")
);


app.use(
bodyParser.json()
);





let writing=false;



function bjTime(){

return new Date()
.toLocaleString(
"zh-CN",
{
timeZone:"Asia/Shanghai"
}
);

}





function loadOrders(){


try{


return JSON.parse(

fs.readFileSync(
DB_FILE,
"utf8"
)

);


}catch(e){


return [];

}


}





async function saveOrders(data){


while(writing){

await new Promise(
r=>setTimeout(r,50)
);

}


writing=true;


try{


fs.writeFileSync(

DB_FILE,

JSON.stringify(
data,
null,
2
)

);


}finally{


writing=false;


}


}








function getIP(req){


return (

req.headers["x-forwarded-for"]

||

req.socket.remoteAddress

||

"unknown"

)

.replace(
"::ffff:",
""
);


}






function isWechat(req){


let ua=

(req.headers["user-agent"]||"")

.toLowerCase();


return ua.includes(
"micromessenger"
);


}







async function getAddress(lat,lng){


try{


let r=await axios.get(

"https://restapi.amap.com/v3/geocode/regeo",

{

params:{

key:
process.env.AMAP_WEB_KEY,

location:
`${lng},${lat}`,

extensions:"base",

output:"JSON"

}

}

);



if(
r.data.status==="1"
){


return r.data

.regeocode

.formatted_address;


}


}catch(e){


console.log(
"高德解析失败:",
e.message
);


}


return "未知位置";


}







async function sendTelegram(text){


if(

!process.env.BOT_TOKEN ||

!process.env.TG_CHAT_ID

){


return;


}



try{


await axios.post(

`https://api.telegram.org/bot${process.env.BOT_TOKEN}/sendMessage`,

{

chat_id:
process.env.TG_CHAT_ID,


text:text,


disable_web_page_preview:true


}

);


}catch(e){


console.log(
"Telegram失败:",
e.message
);


}


}







async function sendEmail(text){


if(

!process.env.EMAIL_USER ||

!process.env.EMAIL_PASS ||

!process.env.EMAIL_TO

){


return;


}




try{


let transporter=

nodemailer.createTransport({

host:"smtp.qq.com",

port:465,

secure:true,


auth:{


user:
process.env.EMAIL_USER,


pass:
process.env.EMAIL_PASS


}


});





await transporter.sendMail({

from:
`智能挪车 <${process.env.EMAIL_USER}>`,


to:
process.env.EMAIL_TO,


subject:
"🚗 新挪车提醒",


text:text


});


}catch(e){


console.log(
"邮箱失败:",
e.message
);


}


}

// 提交订单


app.post(
"/notify",

async(req,res)=>{


let {

lat,

lng,

device,

car

}=req.body;




if(!lat || !lng){


return res.json({

success:false,

message:"定位失败"

});


}





let ip=getIP(req);



let orders=loadOrders();




let old=

orders.find(

o=>

o.ip===ip

&&

o.status!=="finish"

);





if(old){


return res.json({

success:true,

repeat:true,

id:old.id,

address:old.address,

status:old.status,

message:"已经提交，请等待"

});


}






let address=

await getAddress(
lat,
lng
);





let order={


id:uuidv4(),


car:car||"",


ip,


wechat:isWechat(req),


lat,


lng,


address,


status:"waiting",


ownerReceived:false,


receivedTime:"",


createTime:bjTime()


};





orders.unshift(order);



await saveOrders(
orders
);






let url=

`${process.env.PUBLIC_URL}/received.html?id=${order.id}`;





let text=

`🚗 挪车提醒

📍位置：
${address}

🕒时间：
${order.createTime}

🆔订单：
${order.id}

👉确认：
${url}`;





sendTelegram(text);

sendEmail(text);






res.json({

success:true,

id:order.id,

address,

status:order.status

});


}

);

${d.receivedTime||""}

}



if(d.status==="finish"){

`

⚪ 已完成

<br><br>

${d.finishTime||""}

`;

}


}





load();
